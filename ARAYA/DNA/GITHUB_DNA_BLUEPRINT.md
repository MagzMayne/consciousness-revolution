# QUICK COMMANDS

```bash
# === STATUS ===
git status                        # Local status
git remote -v                     # Show remotes
git log --oneline -10             # Recent commits

# === MCP OPERATIONS (Claude CLI) ===
# Search repos
mcp__github__search_repositories query="user:overkillkulture"

# List commits
mcp__github__list_commits owner="overkillkulture" repo="araya-discord-bot"

# Create issue
mcp__github__create_issue owner="overkillkulture" repo="REPO" title="TITLE" body="BODY"

# Get file contents
mcp__github__get_file_contents owner="overkillkulture" repo="REPO" path="README.md"

# Push files
mcp__github__push_files owner="overkillkulture" repo="REPO" branch="main" files=[...] message="MSG"

# === GIT CLI ===
git clone git@github.com:overkillkulture/REPO.git
git push origin main
git pull origin main
git branch -a                     # List all branches

# === ADVANCED ===
git config --global user.signingkey KEYID  # Set GPG key
git commit -S -m "message"                  # Signed commit
gh repo archive overkillkulture/REPO        # Archive repo (gh CLI)
```
