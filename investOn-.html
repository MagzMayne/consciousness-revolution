Bootstrap A — Minimal (bootstrap-minimal.sh)
#!/usr/bin/env bash set -euo pipefail REPO_URL="https://github.com/barbrickdesign/investO.git" APP_DIR="/opt/investo" echo "==> Installing base dependencies (Docker, Git)..." if ! command -v docker &>/dev/null; then sudo apt-get update sudo apt-get install -y docker.io fi if ! command -v docker-compose &>/dev/null; then sudo apt-get install -y docker-compose fi if ! command -v git &>/dev/null; then sudo apt-get install -y git fi echo "==> Cloning or updating repo..." if [ -d "$APP_DIR/.git" ]; then cd "$APP_DIR" git pull --ff-only else sudo mkdir -p "$APP_DIR" sudo chown "$(whoami)":"$(whoami)" "$APP_DIR" git clone "$REPO_URL" "$APP_DIR" cd "$APP_DIR" fi echo "==> Creating .env (if missing)..." if [ ! -f .env ]; then cat > .env <<EOF PAYPAL_CLIENT_ID=REPLACE_ME PAYPAL_SECRET=REPLACE_ME PUMPFUN_PRIVATE_KEY=REPLACE_ME SOLANA_RPC=https://api.mainnet-beta.solana.com POSTGRES_PASSWORD=investo_pass EOF echo "Created .env with placeholders. Sync & failover (sync-and-failover.sh)
#!/usr/bin/env bash set -euo pipefail APP_DIR="/opt/investo" HEALTH_PATH="/api/health" PREFERRED_MODE="${1:-production}" # minimal | production | cloud cd "$APP_DIR" function health_check() { local mode="$1" local port="$2" local url="http://127.0.0.1:${port}${HEALTH_PATH}" curl -fsS "$url" >/dev/null 2>&1 } echo "==> Syncing repo..." git pull --ff-only echo "==> Using mode: ${PREFERRED_MODE}" case "$PREFERRED_MODE" in minimal) docker_compose="docker-compose.minimal.yml" backend_port=8080 ;; production) docker_compose="docker-compose.production.yml" backend_port=8080 ;; cloud) docker_compose="docker-compose.cloud.yml" backend_port=8080 ;; *) echo "Unknown mode: $PREFERRED_MODE" exit 1 ;; esac echo "==> Ensuring .env exists..." if [ ! -f .env ]; then echo "No .env found. Run one of the bootstrap scripts first." exit 1 fi echo "==> Starting preferred stack (${PREFERRED_MODE})..." docker-compose -f "$docker_compose" up -d --build echo "==> Waiting a few seconds for services to come up..." sleep 10 echo "==> Health-checking backend on port ${backend_port}..." if health_check "$PREFERRED_MODE" "$backend_port"; then echo "✅ ${PREFERRED_MODE} stack healthy." exit 0 fi echo "❌ ${PREFERRED_MODE} stack unhealthy. Trying failover..." declare -a modes=("production" "cloud" "minimal") for mode in "${modes[@]}"; do [ "$mode" = "$PREFERRED_MODE" ] && continue echo "==> Trying failover to ${mode}..." case "$mode" in minimal) docker_compose="docker-compose.minimal.yml" ;; production) docker_compose="docker-compose.production.yml" ;; cloud) docker_compose="docker-compose.cloud.yml" ;; esac docker-compose -f "$docker_compose" up -d --build sleep 10 if health_check "$mode" 8080; then echo "✅ Failover successful. ${mode} stack is healthy." exit 0 else echo "❌ ${mode} stack also unhealthy. Trying next..." fi done echo "🚨 All stacks unhealthy. Manual intervention required." exit 1
