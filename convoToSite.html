    // UI bindings
    $("#parseBtn").addEventListener("click", async () => {
      const text = $("#inputText").value.trim();
      if (!text) return alert("Paste conversation JSON or Markdown first.");
      const conv = parseInput(text);
      const cls = classifyConversation(conv);

      const statusEl = $("#classifyStatus");
      statusEl.className = "status " + (
        cls.status === "application-ready" ? "ok" :
        (cls.status === "prototype" || cls.status === "needs modeling") ? "warn" : "err"
      );
      statusEl.innerHTML = `
        Detected → ${cls.status} (score: ${cls.score}/5)
        problem:${cls.hasProblem ? "✓":"×"}
        solution:${cls.hasSolution ? "✓":"×"}
        workflow:${cls.hasWorkflow ? "✓":"×"}
        entities:${cls.hasEntities ? "✓":"×"}
        audience:${cls.hasAudience ? "✓":"×"}
      `;
      await audit.log("classify", { status: cls.status, score: cls.score });

      addConversation(conv, cls);
      renderIndexPage();
      renderNavigation();
      location.hash = "#/pages/index";
      renderPreviewRoute();
      renderContentPanel();
    });
