## DATABASE ADDITIONS FOR GAMIFICATION

```sql
-- Add to users table
ALTER TABLE users ADD COLUMN xp_total INT DEFAULT 0;
ALTER TABLE users ADD COLUMN current_level INT DEFAULT 1;

-- Discovery tracking
CREATE TABLE user_discoveries (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  domain_number INT CHECK (domain_number BETWEEN 1 AND 7),
  level_type VARCHAR(20), -- 'personal', 'team', 'public'
  discovery_content TEXT,
  completed_at TIMESTAMPTZ,
  xp_earned INT,
  UNIQUE(user_id, domain_number, level_type)
);

-- Badges
CREATE TABLE user_badges (
  id UUID PRIMARY KEY,
  user_id UUID REFERENCES users(id),
  badge_code VARCHAR(50),
  earned_at TIMESTAMPTZ
);

-- Leaderboards (materialized view, refresh hourly)
CREATE MATERIALIZED VIEW leaderboard_revenue AS
SELECT
  user_id,
  username,
  SUM(seller_payout_usd) as total_revenue,
  RANK() OVER (ORDER BY SUM(seller_payout_usd) DESC) as rank
FROM transactions t
JOIN users u ON t.seller_user_id = u.id
GROUP BY user_id, username;
```
