    async function exportZip() {
      const files = {
        "index.html": buildExportIndex(),
        "assets/style.css": "/* portable styles */\\nbody{font-family:system-ui;background:#0b0f1a;color:#e5e7eb}",
        "conversations/manifest.json": JSON.stringify(site.manifest, null, 2),
        ...site.files
      };
      const blob = await createZip(files);
      const a = document.createElement("a");
      a.download = ( qs("#repoName").value || "conversation-site" ) + ".zip";
      a.href = URL.createObjectURL(blob);
      a.rel = "noopener";
      a.click();
      await audit.log("export-zip", { files: Object.keys(files).length });
    }

    // UI event bindings (delegated, no inline)
    qs("#parseBtn").addEventListener("click", async () => {
      const text = qs("#inputText").value.trim();
      if (!text) { qs("#classifyStatus").innerHTML = "Paste conversation first."; return; }
      const conv = parseInput(text);
      const cls = classifyConversation(conv);

      const statusEl = qs("#classifyStatus");
      statusEl.className = "status " + (
        cls.status === "application-ready" ? "ok" :
        (cls.status === "prototype" || cls.status === "needs modeling") ? "warn" : "err"
      );
      statusEl.innerHTML = `
        Detected → ${escapeHTML(cls.status)} (score: ${cls.score}/5)
        problem:${cls.hasProblem ? "✓":"×"}
        solution:${cls.hasSolution ? "✓":"×"}
        workflow:${cls.hasWorkflow ? "✓":"×"}
        entities:${cls.hasEntities ? "✓":"×"}
        audience:${cls.hasAudience ? "✓":"×"}
      `;
      await audit.log("classify", { status: cls.status, score: cls.score });

      addConversation(conv, cls);
      renderIndexPage();
      renderNavigation();
      location.hash = "#/pages/index";
      renderPreviewRoute();
      renderContentPanel();
    });

    qs("#sitePreview").addEventListener("click", async (e) => {
      const btn = e.target.closest("button[data-action]");
      if (!btn) return;
      const action = btn.getAttribute("data-action");
      const topic = btn.closest("#actionGrid")?.getAttribute("data-topic") || "Conversation";
      if (action === "preview-app") {
        await audit.log("preview-app", { topic });
        // Non-invasive notification
        console.info("Scaffold preview prepared:", topic);
      } else if (action === "generate-todo") {
        await audit.log("generate-todo", { topic });
        const todo = [
          "- Define entities and relationships",
          "- Implement logging and audit trail",
          "- Build UI components and navigation",
          "- Connect OPFS persistence and Service Worker",
          "- Add monetization toggles (freemium, marketplace)",
          "- Write tests for workflows"
        ].join("\n");
        const slug = slugify(topic);
        site.files[`notes/${slug}-todo.md`] = `# TODO: ${topic}\n\n${todo}\n`;
        renderContentPanel();
      } else if (action === "publish-page") {
        await audit.log("publish-page", { topic });
      }
    });
